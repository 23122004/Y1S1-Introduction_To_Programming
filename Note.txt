* Đoạn code giới hạn ký tự
#include "library.h"
using namespace std;
void gotoxy(int x, int y) // đi đến vị trí x, y trên console
{
	COORD coord;
	coord.X = x;
	coord.Y = y;
	SetConsoleCursorPosition(GetStdHandle(STD_OUTPUT_HANDLE), coord);
}

void curr_coordinate(int &x, int &y) // lấy vị trí hiện tại của con trỏ
{
    CONSOLE_SCREEN_BUFFER_INFO curr;
    GetConsoleScreenBufferInfo(GetStdHandle(STD_OUTPUT_HANDLE), &curr);
    x = curr.dwCursorPosition.X;
    y = curr.dwCursorPosition.Y;
}

int main()
{
    char ch = '`';
    string a; // lưu xâu vào a

    while(1)
    {
        int x, y;
        ch = getch();
        if(ch == 13 || ch == 10) // kiểm tra nếu nhấn enter thì dừng
        {
            printf("\n");
            break;
        }
        if(ch == '\b' && a.size()) // nếu là ký tự backspace thì xóa
        {
            a.pop_back();
            curr_coordinate(x, y);
            x--;
            gotoxy(x, y);
            cout << " ";
            curr_coordinate(x, y);
            x--;
            gotoxy(x, y);
        }
        else if(a.size() < 10) // giới hạn 10 ký tự
        {
            a.push_back(ch);
            cout << ch;
        }
    }
    cout << a; // a hiện tại là tên
    return 0;
}
